  minFreq = max(minFreq,mp[nums[j]]);

                if(minFreq%2==0 && j-i+1>=3*(minFreq/2)){
                    // cout<<i<<" "<<j<<" "<<minFreq<<"\n";
                    bool flag = true,f=false,f2=false;

                    for(auto it:mp){
                        int x = it.second;
                        if(!(x==minFreq/2 || x==minFreq)){
                            flag = false;
                            break;
                        }
                        if(x==minFreq)f=true;
                        else if(x==minFreq/2)f2 = true;
                    }
                    if(flag && f2 && f)maxi = max(maxi,j-i+1);
                }

                if(mp.size()==1){
                    maxi = max(maxi,mp[nums[j]]);
                }
            }
        }
        return maxi;
    }
