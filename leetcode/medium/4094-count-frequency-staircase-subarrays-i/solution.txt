class Solution {
public:
    int countStaircase(vector<int>& nums) {
        int cnt = 0, n = nums.size();

        for(int i = 0; i < n; i++){
            cnt++;
            unordered_map<int,int> mp;
            vector<int> order;

            mp[nums[i]]++;
            order.push_back(nums[i]);

            for(int j = i + 1; j < n; j++){
                if(!mp.count(nums[j])){
                    order.push_back(nums[j]);
                }
                mp[nums[j]]++;

                int size = mp.size();

                if(size * (size + 1) / 2 == j - i + 1){
                    bool flag = true;

                    for(int k = 0; k < size; k++){
                        if(mp[order[k]] != k + 1){
                            flag = false;
                            break;
                        }
                    }

                    if(flag) cnt++;
                }
            }
        }
