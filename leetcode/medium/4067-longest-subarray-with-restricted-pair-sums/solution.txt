    int maxSubarray(vector<int>& nums) {
        unordered_map<int,int>mp;
        int l=0,n=nums.size(),maxi=2;

        if(n <= 2)return n;

        mp[nums[0]]+=1;
        mp[nums[1]]+=1;

        for(int r=2;r<n;r++){
            mp[nums[r]]++;

            while(l<r && fxn(l,r,nums,mp)){
                mp[nums[l]]--;
                if(mp[nums[l]]==0)mp.erase(nums[l]);
                l++;
            }
            maxi = max(maxi,r-l+1);
        }
        return maxi;
    }
