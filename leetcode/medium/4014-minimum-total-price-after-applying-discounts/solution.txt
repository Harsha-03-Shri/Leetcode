class Solution {
public:
    double minPrice(vector<int>& prices, vector<int>& discounts) {
        sort(prices.begin(),prices.end(),greater<int>());
        sort(discounts.begin(),discounts.end(),greater<int>());

        int j = 0;
        double sum=0;

        for(auto it:prices){
            if(j < discounts.size()){
                sum += (1.0*it*(100-discounts[j]))/100;
            }
            else sum += it;
            j++; 
        }
        return sum;
    }
};
