   }
        } 
        // cout<<idx<<"\n";
        for(int i=0;i<idx;i++){
            if(nums[i]!=target[i])return 0;
        }

        // for(auto it:mp2)cout<<it.first<<":"<<it.second<<" ";
        if(mp1 != mp2)return sum;
        for(int i=idx;i<n;i++){
            sum++;
            if(nums[i]!=target[i])return sum;
            mp1[nums[i]]--;

        }
