ector<int>& nums,int i,vector<vector<int>>& res,vector<int>arr){
        if(i >= nums.size()){
            res.push_back(arr);
            return ;
        }
        // while(i+1<nums.size() && nums[i]==nums[i+1])i++;
        arr.push_back(nums[i]);
        fxn(nums,i+1,res,arr);
        arr.pop_back();
        fxn(nums,i+1,res,arr);
        return ;
    }
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>>res;
        fxn(nums,0,res,{});
        return res;
    }
};
