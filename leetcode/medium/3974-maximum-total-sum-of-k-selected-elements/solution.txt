class Solution {
public:
    typedef long long ll;
    long long maxSum(vector<int>& nums, int k, int mul) {
        ll sum = 0;
        sort(nums.begin(),nums.end(),greater<int>());

        for(int i=0;i<k;i++){
            if(mul > 0){
                sum += 1LL*nums[i]*mul;
                mul--;
            }
            else sum += 1LL*nums[i];
        }
        return sum;
    }
};
