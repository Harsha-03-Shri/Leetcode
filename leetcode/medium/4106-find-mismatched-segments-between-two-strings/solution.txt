class Solution {
public:
    vector<vector<string>> findMismatchedSegments(string s1, string s2) {
        vector<vector<string>> res;
        string xs1 = "",xs2="";

        int n = s1.length();
        for(int i=0;i<n;i++){

            if(s1[i]!=s2[i]){
                xs1 += s1[i];
                xs2 += s2[i];
            }
            else{
                if(!xs1.empty()){
                    res.push_back({xs1,xs2});
                    xs1 = "";
                    xs2 = "";
                }
            }
        }
        if(!xs1.empty()){
            res.push_back({xs1,xs2});
        }
        return res;
    }
};
