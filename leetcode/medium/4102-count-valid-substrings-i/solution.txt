class Solution {
public:
    typedef long long ll;
    long long countValidSubstrings(string s) {

        ll n = s.length(),sum=0,prev=0;
        vector<int>st;

        for(int i=0;i<n-2;i++){
            if(s[i]=='a' && s[i+1]=='b' && s[i+2]=='c'){
                ll len = (i+2-prev);
                sum += len*(len+1)/2 - 1;
                prev=i+1;
            }
        }
        // for(auto it:st)cout<<it<<" ";
        // if(sum > 0)sum+=1;
        if(sum==0)return n*(n+1)/2;
        else {
            // cout<<prev<<" "<<sum;
            ll len = n-prev;
            // cout<<"\n"<<len;
            sum += len*(len+1)/2;
        }
        return sum ;
    }
};
