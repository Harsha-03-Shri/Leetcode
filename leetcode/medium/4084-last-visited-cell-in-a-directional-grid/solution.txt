    vector<int> lastVisitedCell(vector<vector<char>>& grid) {
        int x = 0, y =0,tempX=0,tempY=0;

        while(x>=0 && x<grid.size() && y>=0 && y<grid[0].size() && grid[x][y]!='Z'){
            tempX = x,tempY = y;
            
            char c = grid[x][y];
            grid[x][y]='Z';

            if(c=='R')y+=1;
            else if(c=='L')y-=1;
            else if(c=='U')x-=1;
            else if(c=='D')x += 1;

            if(!(x>=0 && x<grid.size() && y>=0 && y<grid[0].size()) || grid[x][y]=='Z')return {tempX,tempY};
        }
        return {tempX,tempY};
    }
