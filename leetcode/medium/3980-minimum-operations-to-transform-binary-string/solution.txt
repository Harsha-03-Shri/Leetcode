class Solution {
public:
    int minOperations(string s1, string s2) {
        int cnt=0,n=s1.size();
        if(n==1){
            if(s1[0]=='1' && s2[0]=='0')return -1;
            if(s1[0]==s2[0])return 0;
            else return 1;
        }

        for(int i=0;i<n;i++){
            if(s1[i]=='0' && s2[i]=='1')cnt++;

            else if(s2[i]=='0' && s1[i]=='1'){
                if(i < n-1 && s1[i+1]=='1'){
                    cnt++;
                    s1[i+1]='0';
                }
                else cnt+=2;
            }
        }

        return cnt;
    }
};
