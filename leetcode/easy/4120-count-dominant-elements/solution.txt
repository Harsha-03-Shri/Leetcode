class Solution {
public:
    int countDominantElements(vector<int>& nums) {
        int n = nums.size(),sum=0;
        if(n==1)return n;

        for(int i=0;i<n;i++){
            int l = i-1 >=0 ? nums[i-1]:0;
            int r = i+1 < n ? nums[i+1]:0;

            int avg = (l==0 || r==0) ? l+r : (l+r+1)/2;
            cout<<avg<<" ";
            if(avg <= nums[i])sum++;
        }
        return sum;
    }
};
