or<int>& nums) {
        int n = nums.size(),cnt=0;
        unordered_map<int,int>mp;

        for(int i=0;i<n;i++){
            if(!mp.count(nums[i])){
                mp[nums[i]] = i;
                cnt+=1;
            }
            else if(mp.count(nums[i]) && i != mp[nums[i]]+1 && mp[nums[i]] >= 0){
                cnt--;
                mp[nums[i]]=INT_MIN;
            }
            else if(mp[nums[i]]>=0){
                mp[nums[i]]=i;
            }
        }
        return cnt;
    }
};
