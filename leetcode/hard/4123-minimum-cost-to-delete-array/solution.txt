class Solution {
public:
    typedef long long ll;
    long long minCost(vector<int>& nums) {
        ll sum=0,ans = nums[0] + nums[nums.size()-1];
        ll minPos=LLONG_MAX;
        priority_queue<ll>neg;

        if(nums.size()<=3)return nums[0] + nums[nums.size()-1];

        for(int i=1;i<nums.size()-1;i++){
            if(nums[i]<0){
                neg.push(nums[i]);
                sum+=nums[i];
            }
            if(nums[i]>=0){
                minPos = min(1LL*minPos,1LL*nums[i]);
            }
        }
        if(neg.size()==1){
            // cout<<"yes"<<sum+minPos;
            if(sum + minPos <= 0)return ans + sum + minPos;
            else sum = 0;
        }
        else if(neg.size()>1 && neg.size()&1){
            if(minPos + neg.top()<=0)return minPos + ans + sum;
            else sum -= neg.top();
            return ans + sum;
        }
        return sum + ans;
    }
};
